theme.css
Bloc de jetons à coller (Tailwind v4)

À ajouter au @theme inline de src/styles/theme.css. Les couleurs shadcn existantes restent en place ; ces jetons s'y ajoutent.

@theme inline {
  /* brand — le rouge RedMark, intentionnel */
  --color-brand-50:  #FEF2F1;
  --color-brand-100: #F9DCD9;
  --color-brand-600: #E10600;   /* base : boutons, FAB, actif */
  --color-brand-700: #C00500;   /* hover */
  --color-brand-800: #8F0400;   /* texte sur fond clair, active */

  /* surfaces & bordures (alias zinc) */
  --color-canvas:  #FAFAFA;     /* fond d'écran */
  --color-surface: #FFFFFF;     /* cartes */
  --color-subtle:  #F4F4F5;     /* zones inertes, séparateurs de lignes */
  --color-line:    #E4E4E7;     /* bordures 1 px */
  --color-line-strong: #D4D4D8;

  /* texte */
  --color-ink:       #1A1A1A;   /* titres, valeurs */
  --color-body:      #3F3F46;
  --color-muted:     #71717A;   /* sur-titres, légendes */
  --color-faint:     #A1A1AA;   /* dates, compteurs secondaires */

  /* sémantique — deux teintes seulement */
  --color-open:     #E10600;
  --color-resolved: #15803D;
  --color-warn:     #A16207;

  --radius-lg: 0.5rem;   /* champs, boutons */
  --radius-xl: 0.75rem;  /* cartes, panneaux */
}
